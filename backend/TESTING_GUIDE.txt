# PART 7: COST CALCULATION VERIFICATION
# ============================================================================

## Understanding Cost Breakdown:

# For a 5-day trip with 2 travelers and defaults:
# - Meal cost: $60/day/person × 5 days × 2 travelers = $600
# - Accommodation: $100/night/person × 4 nights × 2 travelers = $800
#   (nights = days - 1, so 5 days = 4 nights)
# - Transportation: $20/day/person × 5 days × 2 travelers = $200
# - Activity cost: $0 (no activities provided yet)
# - Subtotal: $1,600
# - Contingency (10%): $160
# - Total: $1,760

## Example calculation verification:
curl -X POST http://localhost:8000/api/trips/validate \
  -H "Content-Type: application/json" \
  -d '{
    "destination": "Test",
    "days": 5,
    "travelers": 2,
    "budget": 2000.0,
    "interests": ["testing"]
  }' | python -m json.tool | grep -A 20 "cost_breakdown"

# ============================================================================
# PART 8: EVALUATION SCORING VERIFICATION
# ============================================================================

## Budget Score (0-100):
# - 100: Cost ≤ 80% of budget
# - 85: Cost 80-90% of budget
# - 70: Cost 90-100% of budget
# - 50: Cost 100-110% of budget
# - 0: Cost > 110% of budget

## Preference Score (0-100):
# - Based on activity categories matching interests
# - No activities provided yet = 50 (neutral)

## Activity Diversity Score (0-100):
# - Based on number of unique categories
# - No activities provided yet = 0

## Overall Score = weighted average:
# (budget × 0.30) + (preferences × 0.25) + (diversity × 0.10) +
# (weather × 0.20) + (travel_efficiency × 0.15)

# PART 12: API RESPONSE STRUCTURE REFERENCE
# ============================================================================

# Success response (200 OK) structure:
{
  "success": bool,
  "message": str,
  "trip_details": {
    "destination": str,
    "days": int,
    "travelers": int,
    "interests": [str],
    "max_daily_travel_minutes": int
  },
  "budget_analysis": {
    "user_budget": float,
    "estimated_total": float,
    "within_budget": bool,
    "over_budget": bool,
    "difference": float,
    "percentage_used": float
  },
  "cost_breakdown": {
    "meal_cost": float,
    "accommodation_cost": float,
    "transportation_cost": float,
    "activity_cost": float,
    "subtotal": float,
    "contingency_fund": float,
    "total_cost": float
  },
  "evaluation_scores": {
    "budget_score": float,
    "preference_score": float,
    "activity_diversity_score": float,
    "weather_score": float,
    "travel_efficiency_score": float,
    "overall_score": float
  },
  "recommendations": {
    "needs_replan": bool,
    "suggestions": [str]
  }
}

# Error response (4xx/5xx) structure:
{
  "detail": str  # Error message
}
